/* ── 후리가나 ──────────────────────────────────────
   단어 사전(FDICT)을 최장일치로 훑어 한자 부분에만 <ruby>를 붙인다.
   예문마다 루비를 손으로 달지 않으므로, 내용을 고쳐도 후리가나가
   저절로 따라온다. 사전에 없는 한자는 kbAudit()이 전부 찾아낸다.   */

const HAN = /[㐀-䶿一-鿿豈-﫿々〇]/;
const isHan = c => c !== "〇" && HAN.test(c);

/* 표기와 읽기를 맞춰, 한자 덩어리마다 읽기를 나눠 준다.
   「落ち着く」+「おちつく」 → 落=お · ち · 着=つ · く
   오쿠리가나가 서로 맞물리는 성질을 이용한다.
   맞물리지 않으면 null 을 돌려주고, 부르는 쪽이 더 짧은 후보로 넘어간다. */
function align(word, reading) {
  const segs = [];
  let i = 0;
  while (i < word.length) {
    const han = isHan(word[i]);
    let j = i;
    while (j < word.length && isHan(word[j]) === han) j++;
    segs.push({ t: word.slice(i, j), han });
    i = j;
  }
  const out = [];
  let r = 0;
  for (let k = 0; k < segs.length; k++) {
    const s = segs[k];
    if (!s.han) {
      if (!reading.startsWith(s.t, r)) return null;
      r += s.t.length;
      out.push({ t: s.t, r: "" });
      continue;
    }
    const next = segs[k + 1];
    let end;
    if (!next) end = reading.length;
    else {
      end = reading.indexOf(next.t, r + 1);
      if (end < 0) return null;
    }
    const rd = reading.slice(r, end);
    if (!rd) return null;
    out.push({ t: s.t, r: rd });
    r = end;
  }
  return r === reading.length ? out : null;
}

const FKEYS = Object.keys(FDICT);
const FMAXLEN = FKEYS.reduce((m, k) => Math.max(m, k.length), 1);
const FSET = new Set(FKEYS);
const rubyCache = new Map();

/* 문자열 안의 한자에 루비를 입힌다. HTML 태그는 건드리지 않는다. */
function ruby(str) {
  if (str == null) return "";
  const s = String(str);
  if (!HAN.test(s)) return s;
  const cached = rubyCache.get(s);
  if (cached !== undefined) return cached;

  let out = "";
  let i = 0;
  while (i < s.length) {
    const c = s[i];
    if (c === "<") {                       // 태그는 통째로 넘긴다
      const gt = s.indexOf(">", i);
      if (gt < 0) { out += s.slice(i); break; }
      out += s.slice(i, gt + 1); i = gt + 1; continue;
    }
    /* 최장일치 — 사전에 있고 읽기까지 맞물리는 가장 긴 후보를 고른다.
       「お得」「お腹」처럼 가나로 시작하는 표제어도 잡아야 하므로,
       한자 자리뿐 아니라 모든 자리에서 사전을 본다. 다만 가나 자리에서는
       한자를 품은 표제어만 인정한다(「する」 같은 것이 걸리지 않도록). */
    const atKana = !isHan(c);
    let pairs = null, hitLen = 0;
    for (let len = Math.min(FMAXLEN, s.length - i); len >= 1; len--) {
      const cand = s.substr(i, len);
      if (!FSET.has(cand)) continue;
      if (atKana && !HAN.test(cand)) continue;
      const p = align(cand, FDICT[cand]);
      if (p) { pairs = p; hitLen = len; break; }
    }
    if (!pairs && atKana) { out += esc(c); i++; continue; }
    if (pairs) {
      pairs.forEach(p => {
        out += p.r ? `<ruby>${esc(p.t)}<rt>${esc(p.r)}</rt></ruby>` : esc(p.t);
      });
      i += hitLen;
      continue;
    }
    out += `<span class="nofuri">${esc(c)}</span>`;   // 사전에 없는 한자
    i++;
  }
  rubyCache.set(s, out);
  return out;
}

/* 감사 — 사전이 못 덮은 한자를 전부 찾아낸다. 콘솔에서 kbAudit(). */
function kbAudit() {
  const miss = new Map();
  let n = 0;
  const visit = (txt, where) => {
    if (typeof txt !== "string" || !HAN.test(txt)) return;
    n++;
    const re = /<span class="nofuri">(.)<\/span>/g;
    const html = ruby(txt);
    let m;
    while ((m = re.exec(html))) {
      const ch = m[1];
      if (!miss.has(ch)) miss.set(ch, []);
      if (miss.get(ch).length < 3) miss.get(ch).push(where + " :: " + txt.slice(0, 44));
    }
  };
  DAYS.forEach(d => {
    visit(d.theme, `${d.no}일 theme`);
    visit(d.root, `${d.no}일 root`);
    visit(d.rootKo, `${d.no}일 rootKo`);
    d.words.forEach((w, i) => {
      visit(w.ja, `${d.no}일 단어${i + 1}`);
      visit(w.note, `${d.no}일 단어${i + 1} 설명`);
      w.exs.forEach((e, j) => visit(e.s, `${d.no}일 단어${i + 1} 예문${j + 1}`));
      if (w.exs.some(e => e.g)) w.exs.forEach((e, j) => visit(e.g, `${d.no}일 단어${i + 1} 문법${j + 1}`));
    });
    d.branches.forEach((b, i) => { visit(b.ja, `${d.no}일 확장${i + 1}`); visit(b.note, `${d.no}일 확장${i + 1} 설명`); });
    d.quiz.forEach((q, i) => { visit(q.q, `${d.no}일 문제${i + 1}`); q.opts.forEach(o => visit(o, `${d.no}일 문제${i + 1} 보기`)); });
    const b = d.bunka;
    visit(b.title, `${d.no}일 상식 제목`); visit(b.vs, `${d.no}일 상식 비교`);
    b.body.concat(b.body2).forEach(p => visit(p, `${d.no}일 상식 본문`));
  });
  const list = [...miss.entries()];
  console.log(`검사 ${n}건 · 못 덮은 한자 ${list.length}종`);
  return { count: list.length, missing: list.map(x => x[0]), detail: Object.fromEntries(list) };
}

/* 실제로 붙은 읽기를 전부 모아 눈으로 검수하기 위한 목록 */
function kbReadings() {
  const used = new Map();
  const scan = txt => {
    if (typeof txt !== "string" || !HAN.test(txt)) return;
    const re = /<ruby>([^<]*)<rt>([^<]*)<\/rt><\/ruby>/g;
    let m; const html = ruby(txt);
    while ((m = re.exec(html))) {
      const k = m[1] + "|" + m[2];
      used.set(k, (used.get(k) || 0) + 1);
    }
  };
  DAYS.forEach(d => {
    visitAll(d, scan);
  });
  return [...used.entries()].map(([k, c]) => { const [w, r] = k.split("|"); return { w, r, c }; })
    .sort((a, b) => a.w.localeCompare(b.w, "ja"));
}
function visitAll(d, f) {
  f(d.theme); f(d.root); f(d.rootKo);
  d.words.forEach(w => { f(w.ja); f(w.note); w.exs.forEach(e => { f(e.s); if (e.g) f(e.g); }); });
  d.branches.forEach(b => { f(b.ja); f(b.note); });
  d.quiz.forEach(q => { f(q.q); q.opts.forEach(f); });
  f(d.bunka.title); f(d.bunka.vs);
  d.bunka.body.concat(d.bunka.body2).forEach(f);
}
if (typeof window !== "undefined") { window.kbAudit = kbAudit; window.kbReadings = kbReadings; }
